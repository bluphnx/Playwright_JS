2. Page Objects (POM Layer)

Each class wraps one page/screen: locators live in the constructor, actions are methods. POManager is a factory that exposes all pages through getters.

LoginPage.js/pageobjects/LoginPage.js
Concept: Page Object Model — locators in constructor, goTo() and ValidLogin() actions exported via module.exports.
class LoginPage {

    constructor(page) {
        this.page = page;
        this.SignInBtn = page.locator("#login");
        this.userName = page.locator("[formcontrolname='userEmail']");
        this.Password = page.locator("#userPassword");
    }

    async goTo() {
        await this.page.goto("https://rahulshettyacademy.com/client");
    }

    async ValidLogin(username, password) {
        await this.userName.fill(username);
        await this.Password.fill(password);
        await this.SignInBtn.click();
    }
}

module.exports = { LoginPage };
DashboardPage.js/pageobjects/DashboardPage.js
Concept: Looping over product cards with allTextContents(), matching a product, clicking the right .nth(index) Add-to-Cart, then a web-first assertion toContainText that auto-waits.
const { expect } = require('@playwright/test');

class DashboardPage {

    constructor(page) {
        this.page = page;
        this.cardTitles = page.locator(".card-body b");
        this.cardPrices = page.locator("div.card-body .text-muted");
        this.allAddToCart = page.locator("button.w-10");
        this.AddedToCartSuccessMsg = page.locator("#toast-container");
    }

    async SearchAndAddToCart(Product) {
        await this.cardTitles.last().waitFor();
        const allTitles = await this.cardTitles.allTextContents();
        const allPrices = await this.cardPrices.allTextContents();

        for (let index = 0; index < allTitles.length; index++) {
            console.log(`${index + 1}) Product : ${allTitles[index]} => Price : ${allPrices[index]}`);

            if (`${allTitles[index]}` == Product) {
                await this.allAddToCart.nth(index).click();
                await this.AddedToCartSuccessMsg.locator("text=Product Added To Cart").waitFor();
                console.log(await this.AddedToCartSuccessMsg.textContent());
                // Option: Use toContainText — auto-waits for the text to appear
                await expect(this.AddedToCartSuccessMsg).toContainText('Product Added To Cart');
                break;
            }
        }
    }
}

module.exports = { DashboardPage };
CartPage.js/pageobjects/CartPage.js
Concept: pressSequentially to trigger an autocomplete dropdown, iterating options with count()/nth(), and placing the order.
const { expect } = require('@playwright/test');

class CartPage {

    constructor(page) {
        this.page = page;
    }

    async NavigateToCart() {
        await this.page.locator("[routerlink*='cart']").click();
        await this.page.locator("div li").last().waitFor();
    }

    async PlaceOrder(email) {
        await this.page.locator("text=Checkout").click();
        await this.page.locator("[placeholder*='Country']").pressSequentially('Ind');

        const dropdownlist = this.page.locator(".ta-results");
        await dropdownlist.waitFor();
        const optionscount = await dropdownlist.locator("button").count();

        for (let i = 0; i < optionscount; ++i) {
            let text = await dropdownlist.locator("button").nth(i).textContent();
            if (text === ' India') {
                await dropdownlist.locator("button").nth(i).click();
                break;
            }
        }

        expect(await this.page.locator(".user__name [type='text']").first()).toHaveText(email);
        await this.page.locator(".action__submit").click();
    }
}

POManager.js/pageobjects/POManager.js
Concept: Factory / Facade pattern — instantiates all page objects once and exposes them via getters, so a test constructs one manager instead of five classes.
const { LoginPage } = require('./LoginPage')
const { DashboardPage } = require('./DashboardPage')
const { CartPage } = require('./CartPage')
const { OrderSummaryPage } = require('./OrderSummaryPage')

class POManager {

    constructor(page) {
        this.page = page;
        this.loginPage = new LoginPage(this.page);
        this.dashboardPage = new DashboardPage(this.page);
        this.cartPage = new CartPage(this.page);

    getDashboardPage() { return this.dashboardPage; }

3. Utilities & Test Data

Reusable helpers that hide low-level detail from the tests: an API facade, two custom-fixture definitions, and JSON data sources.

APIutils.js/utils/APIutils.js
Concept: API Facade — getToken() logs in and returns the JWT; createOrder() reuses that token in the Authorization header and returns { token, orderid }. Tests call one method instead of chaining raw requests.
class APIutils {

    constructor(apicontext, loginPayLoad) {
        this.apicontext = apicontext;
        this.loginPayLoad = loginPayLoad;
    }

    async getToken() {
        // login api
        const loginRespone = await this.apicontext.post("https://rahulshettyacademy.com/api/ecom/auth/login",
            { data: this.loginPayLoad });

        const loginResponeJson = await loginRespone.json();
        const token = loginResponeJson.token;
        console.log(token);
        return token;
    }

    async createOrder(orderPayload) {
        // order api
        let response = {};
        response.token = await this.getToken();

        const orderRespone = await this.apicontext.post("https://rahulshettyacademy.com/api/ecom/order/create-order",
            {
                data: orderPayload,
                headers: {
                    'Authorization': response.token,
                    'Content-Type': 'application/json; charset=utf-8'
                }
            });

        console.log("Body:", await orderRespone.text());

        const orderResponeJson = await orderRespone.json();
        const ordersID = orderResponeJson.orders[0];
        console.log('orderResponeJson.orders[0]: ' + ordersID);
        const productOrderId = orderResponeJson.productOrderId[0];
        console.log('orderResponeJson.productOrderId[0]: ' + productOrderId);
        const orderResponeMessage = orderResponeJson.message;
        console.log('orderResponeJson.message: ' + orderResponeMessage);

        response.orderid = ordersID;
        return response;
    }
}

module.exports = { APIutils };

webapi_network_route.spec.js/tests/webapi_network_route.spec.js
Concept — Network Interception / Response Mocking (page.route + route.fulfill): This is the real network-interception example. It intercepts the "get orders for customer" API, fetches the genuine response with page.request.fetch(route.request()), then overrides the body with a fake payload ({ data: [], message: "No Orders" }) via route.fulfill. The * wildcard makes the pattern match any customer id. page.waitForResponse(...) (like waitForLoadState('networkidle')) prevents the assertion from reading a stale message before the mocked response renders.
const { test, expect, request } = require("playwright/test");
const { APIutils } = require("../utils/APIutils");

let apicontext;
const loginPayLoad = { userEmail: "r123am@gmail.com", userPassword: "Amazon@123" };
const orderPayload = { orders: [{ country: "Vietnam", productOrderedId: "6960ea76c941646b7a8b3dd5" }] }; // iphone 13 pro
let response;
let FakePayLoadOrder = { data: [], message: "No Orders" };   // fake body we will inject

test.beforeAll(async () => {
    apicontext = await request.newContext();
    const apiutils = new APIutils(apicontext, loginPayLoad);
    response = await apiutils.createOrder(orderPayload);

test("@API E2E - webapi - part 1", async ({ browser }) => {
    const context = await browser.newContext();
    const page = await context.newPage();

    // inject token so the UI opens logged-in
    await page.addInitScript((value) => {
        window.localStorage.setItem("token", value);
    }, response.token);

    const OpenUrl = page.goto("https://rahulshettyacademy.com/client");
    const myOrdersTab = page.locator("button[routerlink*='myorders']");

    await OpenUrl;

    // ===== NETWORK INTERCEPTION =====
    // API sends real response --> |Playwright fakes/overrides it| --> browser renders our fake data
    await page.route(
        "https://rahulshettyacademy.com/api/ecom/order/get-orders-for-customer/*",
        async (route) => {
            // fetch the genuine response first
            const FetchedResponse = await page.request.fetch(route.request());
            // serialize our fake body
            let body = JSON.stringify(FakePayLoadOrder);
            // override the body sent to the browser
            route.fulfill({
                FetchedResponse,
                body,
            });
        },
    );

    await myOrdersTab.click(); // tap orders tab in navbar
    await page.pause();

    // wait for the mocked response so we don't read a stale message
    // waitForResponse behaves like waitForLoadState("networkidle") here
    await page.waitForResponse("https://rahulshettyacademy.com/api/ecom/order/get-orders-for-customer/*");

    // prints the "No Orders" message that came from our fake payload
    console.log(await page.locator(".mt-4").textContent());

    await context.close();
    await browser.close();
});

reversewords.js/Playwright Concepts/reversewords.js
Concept: Classic interview problem — reverse each word in a sentence using functional chaining split → map → reverse → join. "My name is John" → "yM eman si nhoJ".
function reverseEachWord(str) {
    return str
        .split(' ')
        .map(word => word.split('')
            .reverse()
            .join('')
        ).join(' ');
}

reversewords-basic.js/Playwright Concepts/reversewords-basic.js
Concept: The same problem written the long way with explicit loops and step-by-step comments — great for explaining the logic in an interview before showing the one-liner.
function reverseEachWord(str) {
    // Step 1: Split sentence into array of words
    const words = str.split(' ');          // ["My","name","is","John"]
    const reversedWords = [];

    // Step 3: Loop through each word
    for (let i = 0; i < words.length; i++) {
        const word = words[i];
        const characters = word.split('');     // "My" -> ['M','y']
        const reversedChars = characters.reverse();  // -> ['y','M']
        const reversedWord = reversedChars.join(''); // -> "yM"
        reversedWords.push(reversedWord);
    }

    const result = reversedWords.join(' ');   // "yM eman si nhoJ"
    return result;
}

console.log(reverseEachWord("My name is John"));
