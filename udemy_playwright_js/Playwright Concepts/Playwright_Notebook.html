2. Configuration — playwright.config.js

Central control file — browsers, timeouts, reporters, screenshots.

import { defineConfig } from '@playwright/test';

export default defineConfig({
  testDir: './tests',          // Where test files live
  retries: 2,                  // Retry failed tests 2 times
  timeout: 35 * 1000,         // 35 seconds per test
  reporter: 'html',           // Generate HTML report

  expect: {
    timeout: 45 * 1000        // Assertions wait up to 45 seconds
  },

  use: {
    browserName: 'chromium',   // Default browser
    headless: true,            // No visible browser (fast)
    screenshot: 'only-on-failure', // Capture on failure
    trace: 'on'                // Record trace for debugging
  }
});

Key Points:

• testDir — must point to a FOLDER, not a file
• retries: 2 — handles flaky tests automatically
• timeout — if test exceeds this, it fails
• trace: 'on' — records full execution for debugging
• screenshot: 'only-on-failure' — saves space, captures when needed

3. Page Objects — POM Pattern

Each page in the app has its own class. Locators + actions in one file.

LoginPage.js

class LoginPage {
    constructor(page) {
        this.page = page;
        this.SignInBtn = page.locator("#login");
        this.userName = page.locator("[formcontrolname='userEmail']");
        this.Password = page.locator("#userPassword");
    }

    async goTo() {
        await this.page.goto("https://rahulshettyacademy.com/client");
    }

    async ValidLogin(username, password) {
        await this.userName.fill(username);
        await this.Password.fill(password);
        await this.SignInBtn.click();
    }
}
module.exports = { LoginPage };

→ Locators defined in constructor. Actions are methods. Tests just call loginPage.ValidLogin().

DashboardPage.js

class DashboardPage {
    constructor(page) {
        this.page = page;
        this.cardTitles = page.locator(".card-body b");
        this.allAddToCart = page.locator("button.w-10");
        this.AddedToCartSuccessMsg = page.locator("#toast-container");
    }

    async SearchAndAddToCart(Product) {
        await this.cardTitles.last().waitFor();
        const allTitles = await this.cardTitles.allTextContents();

        for (let index = 0; index < allTitles.length; index++) {
            if (allTitles[index] === Product) {
                await this.allAddToCart.nth(index).click();
                await this.AddedToCartSuccessMsg
                    .locator("text=Product Added To Cart").waitFor();
                break;
            }
        }
    }
}
module.exports = { DashboardPage };

→ Loops through products, finds match, clicks Add to Cart, waits for toast confirmation.

CartPage.js

class CartPage {
    constructor(page) { this.page = page; }

    async NavigateToCart() {
        await this.page.locator("[routerlink*='cart']").click();
        await this.page.locator("div li").last().waitFor();
    }

    async PlaceOrder(email) {
        await this.page.locator("text=Checkout").click();
        await this.page.locator("[placeholder*='Country']")
            .pressSequentially('Ind');

        const dropdownlist = this.page.locator(".ta-results");
        await dropdownlist.waitFor();
        const count = await dropdownlist.locator("button").count();

        for (let i = 0; i < count; ++i) {
            let text = await dropdownlist.locator("button").nth(i).textContent();
            if (text === ' India') {
                await dropdownlist.locator("button").nth(i).click();
                break;
            }
        }
        await this.page.locator(".action__submit").click();
    }
}
module.exports = { CartPage };

OrderSummaryPage.js

class OrderSummaryPage {
    constructor(page) {
        this.page = page;
        this.myOrdersLink = page.locator("label[routerlink*='myorders']");
    }

    async verifyOrder() {
        await this.page.locator("h1.hero-primary").waitFor();
        const orderidstring = await this.page.locator("label.ng-star-inserted")
            .textContent();
        let orderid = orderidstring.replaceAll("|", "").trim();

        await this.myOrdersLink.click();
        await this.page.locator("tbody tr th").last().waitFor();
        const orderCount = await this.page.locator("tbody tr th").count();

        for (let index = 0; index < orderCount; index++) {
            if (await this.page.locator("tbody tr th").nth(index)
                .textContent() === orderid) {
                await this.page.locator("tr .btn-primary").nth(index).click();
                break;
            }
        }
        await expect(this.page.locator("div.col-text")).toHaveText(orderid);
    }
}
module.exports = { OrderSummaryPage };

4. PO Manager — Central Hub

Creates all page objects in one place. Tests import only POManager.

const { LoginPage } = require('./LoginPage');
const { DashboardPage } = require('./DashboardPage');
const { CartPage } = require('./CartPage');
const { OrderSummaryPage } = require('./OrderSummaryPage');

class POManager {
    constructor(page) {
        this.page = page;
        this.loginPage = new LoginPage(this.page);
        this.dashboardPage = new DashboardPage(this.page);
        this.cartPage = new CartPage(this.page);
        this.orderSummaryPage = new OrderSummaryPage(this.page);
    }

    getLoginPage() { return this.loginPage; }
    getDashboardPage() { return this.dashboardPage; }
    getCartPage() { return this.cartPage; }
    getOrderSummaryPage() { return this.orderSummaryPage; }
}
module.exports = { POManager };

Usage in test:

const poManager = new POManager(page);
const loginPage = poManager.getLoginPage();
const dashboardPage = poManager.getDashboardPage();

→ One import, access all pages. Change page object internals without touching tests.

const base = require('@playwright/test');

exports.customtest = base.test.extend({
    authenticatedPage: async ({ browser }, use) => {
        const context = await browser.newContext();
        const page = await context.newPage();
        const email = "r123am@gmail.com";

6. Tests — Basic (Fixtures, Navigation)

const { test, expect } = require('@playwright/test');

// Page fixture — simplest test
test('Navigate to Google', async ({ page }) => {
    await page.goto("https://www.google.com");
    await expect(page).toHaveTitle('Google');
});

// Browser fixture — manual context control
test('Browser context test', async ({ browser }) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    await page.goto("https://www.google.com");
    await expect(page).toHaveTitle('Google');
    await context.close();
});

→ { page } = Playwright manages browser. { browser } = YOU manage context + cleanup.

8. Tests — E2E (Full Order Flow with POM)

const { test } = require('@playwright/test');
const { POManager } = require('../pageobjects/POManager');

test('E2E - place order', async ({ browser }) => {
    const context = await browser.newContext();
    const page = await context.newPage();
    const poManager = new POManager(page);

    const loginPage = poManager.getLoginPage();
    const dashboardPage = poManager.getDashboardPage();
    const cartPage = poManager.getCartPage();
    const orderSummaryPage = poManager.getOrderSummaryPage();

    await loginPage.goTo();
    await loginPage.ValidLogin("r123am@gmail.com", "Amazon@123");
    await dashboardPage.SearchAndAddToCart("ZARA COAT 3");
    await cartPage.NavigateToCart();
    await cartPage.PlaceOrder("r123am@gmail.com");
    await orderSummaryPage.verifyOrder();

    await context.close();
});

→ Clean test! No locators, just method calls. POM handles everything internally.

9. Tests — API Testing

const { test, expect, request } = require('@playwright/test');

let token;

test.beforeAll(async () => {
    const apiContext = await request.newContext();
    const loginResponse = await apiContext.post(
        "https://rahulshettyacademy.com/api/ecom/auth/login",
        { data: { userEmail: "r123am@gmail.com", userPassword: "Amazon@123" } }
    );
    expect(loginResponse.ok()).toBeTruthy();
    const json = await loginResponse.json();
    token = json.token;
});

test('API + UI combined', async ({ page }) => {
    // Inject token into browser (skip UI login!)
    await page.addInitScript(value => {
        window.localStorage.setItem('token', value);
    }, token);

    await page.goto("https://rahulshettyacademy.com/client");
    // Page loads already authenticated!
});

→ Login via API (200ms) instead of UI (3-5s). Inject token with addInitScript. 10x faster.

10. Tests — BDD Cucumber

Feature File (ecommerce.feature)

Feature: Ecommerce Validation
    @E2E
    Scenario: Place order and verify
        Given I login to the ecommerce application with "r123am@gmail.com" and "Amazon@123"
        When I search for "ADIDAS ORIGINAL" and add it to cart
        And I verify product is displayed in the cart
        And I enter account details and place the order for the "r123am@gmail.com"
        Then I should see the order in the order history page

Step Definitions (steps.js)

const { Given, When, Then, setDefaultTimeout } = require('@cucumber/cucumber');
const { expect } = require('@playwright/test');
setDefaultTimeout(30000);

Given('I login to the ecommerce application with {string} and {string}',
    async function(username, password) {
    const loginPage = new LoginPage(this.page);
    await loginPage.goTo();
    await loginPage.ValidLogin(username, password);
});

When('I search for {string} and add it to cart',
    async function(product) {
    const dashboardPage = new DashboardPage(this.page);
    await dashboardPage.SearchAndAddToCart(product);
});

Hooks (hooks.js)

const { After, Before } = require('@cucumber/cucumber');
const playwright = require('playwright');

Before(async function() {
    this.browser = await playwright.chromium.launch({ headless: true });
    this.context = await this.browser.newContext();
    this.page = await this.context.newPage();
});

After(async function() {
    await this.page.close();
    await this.context.close();
    await this.browser.close();
});

→ Run with: npx cucumber-js --tags "@E2E"
