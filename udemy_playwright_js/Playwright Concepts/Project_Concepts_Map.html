1. playwright.config.js — Configuration

TimeoutsRetriesReporterScreenshotsTrace 
export default defineConfig({
  testDir: './tests',
  retries: 2,                        // Retry failed tests 2x
  timeout: 35 * 1000,               // 35s per test
  reporter: 'html',                  // HTML report
  expect: { timeout: 45 * 1000 },   // Assertions: 45s
  use: {
    browserName: 'chromium',
    headless: true,
    screenshot: 'only-on-failure',
    trace: 'on'
  }
});

Concepts:

  • testDir — must be a FOLDER path
  • retries — auto-rerun failed tests
  • timeout — max time per test (ceiling)
  • expect.timeout — assertion retry duration
  • trace: 'on' — records DOM snapshots for debugging
  • screenshot: 'only-on-failure' — saves on error

LoginPage.js — Basic POM

class LoginPage {
    constructor(page) {
        this.page = page;
        this.userName = page.locator("[formcontrolname='userEmail']");
        this.Password = page.locator("#userPassword");
        this.SignInBtn = page.locator("#login");
    }
    async goTo() { await this.page.goto("https://..."); }
    async ValidLogin(u, p) {
        await this.userName.fill(u);
        await this.Password.fill(p);
        await this.SignInBtn.click();
    }
}

APIutils.js — Reusable API Helper Class

class APIutils {
    constructor(apicontext, loginPayLoad) { ... }
    async getToken() {
        const resp = await this.apicontext.post('/login', { data: this.loginPayLoad });
        return (await resp.json()).token;
    }
    async createOrder(orderPayload) {
        const token = await this.getToken();
        const resp = await this.apicontext.post('/create-order', {
            data: orderPayload,
            headers: { 'Authorization': token }
        });
        return { token, orderid: (await resp.json()).orders[0] };
    }
}

steps.js — Step Definitions

const { Given, When, Then, setDefaultTimeout } = require('@cucumber/cucumber');
setDefaultTimeout(30000);  // Increase from 5s default!

Given('I login with {string} and {string}', async function(user, pass) {
    const loginPage = new LoginPage(this.page);  // 'this' = World context
    await loginPage.goTo();
    await loginPage.ValidLogin(user, pass);
});

hooks.js — Lifecycle Management

Before(async function() {
    this.browser = await playwright.chromium.launch({ headless: true });
    this.context = await this.browser.newContext();
    this.page = await this.context.newPage();
});

After(async function() {
    await this.browser.close();  // MUST close or process hangs!
});

AfterStep(async function({ result }) {
    if (result.status === Status.FAILED) {
        await this.page.screenshot({ path: 'failure.png' });
    }
});
