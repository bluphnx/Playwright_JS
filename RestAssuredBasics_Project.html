RestAssuredBasics

A consolidated reference of all source files, POJOs, resources, and notes from the REST Assured API automation learning project.

Project Overview

This project is a hands-on learning workspace for API test automation using REST Assured against a ServiceNow developer instance. It progresses from basic GET calls, through path/query parameters and POST request bodies, into POJO serialization/deserialization with Jackson, Gson, jsonschema2pojo, and Apache Juneau.

Key concepts covered: Static imports of RestAssured (given/when/then), Basic Auth, content-type & accept headers, path parameters, query parameters, request bodies (String, File, Object), Hamcrest assertions (status code, status line, headers, response time, body), POJO serialization with Jackson vs Gson (null handling), jsonschema2pojo build plugin, and Apache Juneau JSON/XML/HTML serialization.

package session_5;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;

import static io.restassured.RestAssured.*;

import org.hamcrest.Matcher;
import org.hamcrest.Matchers;

public class Get_all_Hardware_Category_using_QueryParameter {

	public static void main(String[] args) {

String baseURI = "https://dev230932.service-now.com";
String basePath = "/api/now/table/";
String table_Name = "incident";
String category = "Software";
		
		RestAssured
		.given()
		.log().all()
		.baseUri(baseURI)
		.basePath(basePath)
		.contentType(ContentType.JSON)
//		the below statement returns response in xml format when accept header is set as "application/xml"
//		.accept("application/xml")
		.accept("application/json")
		.auth()
		.basic("admin", "ZB^rjmGvW9%6")
		.pathParam("tableName", table_Name)
//		to filter based on some condition few records from the table - we use query parameter (queryParam)
//		to extract one single record from the table - we use path parameter (pathParam)
		.queryParam("category", category)
		.when()
//		same name provided in the path parameter should be called inside get call as {tableName} , {sys_id}
//		if we call with wrong string we will throw error {sysID} , {table_Name}
		.get("/{tableName}")
		.then()
		.log().all()
		.assertThat()
//		validates response status code is successful - 200
		.statusCode(200)
		.statusLine(Matchers.containsString("OK"))
		.statusLine(Matchers.equalTo("HTTP/1.1 200 OK"))  
		.header("Content-Type", Matchers.containsString("/json"))
		.header("Content-Type", Matchers.equalTo("application/json;charset=UTF-8"))
//		3000L refers to the response should be received less that 3 seconds
		.time(Matchers.lessThan(3000L)); 
		
	}

}

	// as per pojo definition , we need to create the variable with private modifier; 
	
	// can we create with public modifier?
	// yes but best practice is to use private modifier and assign value to variable using setter methods

	// for all post, put, patch methods we use pojo serialization to assign values since the json will be very large and
	// we have to assign vales using serialization

	private String description;
	private String short_description;
	private String category;

	}

import org.apache.juneau.annotation.Beanc;

public class ProductPojoClass {
	
	private String name;
	private String color;
	private int price;
	private String[] sellerName;
	
	// beanc properties not recogonized in constructor
	// so we need to add @Beanc(properties) above constructor
	// to make it work
	// import org.apache.juneau.annotation.Beanc;
	// then add @Beanc(properties="name,color,price,sellerName")
	// above constructor
	// this will tell juneau to use these properties in this order
	
	@Beanc(properties="name,color,price,sellerName")
	public ProductPojoClass(String name, String color, int price, String[] sellerName) {
		super();
		this.name = name;
		this.color = color;
		this.price = price;
		this.sellerName = sellerName;
	}

		return name;
	}

		this.name = name;
	}

		return color;

		this.color = color;
	}
	public int getPrice() {
		return price;
	}
	public void setPrice(int price) {
		this.price = price;
	}
	public String[] getSellerName() {
		return sellerName;
	}
	public void setSellerName(String[] sellerName) {
		this.sellerName = sellerName;
	}
